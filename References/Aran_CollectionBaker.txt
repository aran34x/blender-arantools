import bpy
from collections import Counter

# ------------------------------------------------------------------------
#    Data Properties
# ------------------------------------------------------------------------

class BakeObjectProperties(bpy.types.PropertyGroup):
    target_name: bpy.props.StringProperty(
        name="Target Name",
        description="Meshes with the same Target Name will be joined together.",
        default=""
    )

class BakeSceneProperties(bpy.types.PropertyGroup):
    source_collection: bpy.props.PointerProperty(
        name="Source Collection",
        type=bpy.types.Collection
    )
    target_collection: bpy.props.PointerProperty(
        name="Target Collection",
        type=bpy.types.Collection
    )

# ------------------------------------------------------------------------
#    The Bake Operator (Duplicate -> Convert -> Join)
# ------------------------------------------------------------------------

class OBJECT_OT_SimpleBake(bpy.types.Operator):
    bl_idname = "object.simple_bake"
    bl_label = "Bake & Join"
    bl_description = "Duplicate, Convert to Mesh, Join by Name"
    bl_options = {'REGISTER', 'UNDO'}

    def execute(self, context):
        props = context.scene.simple_bake_props
        source_col = props.source_collection
        target_col = props.target_collection

        if not source_col or not target_col:
            self.report({'ERROR'}, "Please select Source and Target collections.")
            return {'CANCELLED'}
        
        if source_col == target_col:
            self.report({'ERROR'}, "Source and Target cannot be the same.")
            return {'CANCELLED'}

        # ------------------------------------------------------
        # 1. Setup: Clean Target & Ensure Visibility
        # ------------------------------------------------------
        
        # We must ensure both collections are visible for operators to work correctly
        source_col.hide_viewport = False
        target_col.hide_viewport = False
        
        # Clean Target Collection
        for obj in list(target_col.objects):
            bpy.data.objects.remove(obj, do_unlink=True)

        # ------------------------------------------------------
        # 2. Duplicate & Convert (Apply Modifiers)
        # ------------------------------------------------------
        
        # Dictionary to hold our processed objects: {'TargetName': [obj1, obj2]}
        objects_by_name = {}

        # Deselect everything first
        bpy.ops.object.select_all(action='DESELECT')

        for obj in source_col.objects:
            if obj.type != 'MESH':
                continue

            # A. Select ONLY the current object
            obj.select_set(True)
            context.view_layer.objects.active = obj
            
            # B. Duplicate
            bpy.ops.object.duplicate(linked=False)
            new_obj = context.active_object # The duplicate is now active
            
            # C. Move to Target Collection
            # Unlink from all current collections (duplicate stays in source by default)
            for col in new_obj.users_collection:
                col.objects.unlink(new_obj)
            target_col.objects.link(new_obj)
            
            # D. CONVERT TO MESH 
            # This is the "Magic" step that applies all modifiers destructively
            # exactly as if you clicked "Convert to Mesh" in the menu.
            bpy.ops.object.convert(target='MESH')
            
            # E. Store for Sorting
            raw_name = obj.bake_props.target_name.strip()
            key_name = raw_name if raw_name else obj.name
            
            if key_name not in objects_by_name:
                objects_by_name[key_name] = []
            objects_by_name[key_name].append(new_obj)
            
            # Deselect to prepare for next loop
            bpy.ops.object.select_all(action='DESELECT')

        # ------------------------------------------------------
        # 3. Join & Rename
        # ------------------------------------------------------

        for key_name, obj_list in objects_by_name.items():
            if not obj_list:
                continue

            final_name = f"{key_name}_Baked"

            # Select all objects in this group
            for o in obj_list:
                o.select_set(True)
            
            # Set Active Object (Parent of the join)
            context.view_layer.objects.active = obj_list[0]
            
            # Join if more than one
            if len(obj_list) > 1:
                bpy.ops.object.join()
            
            # Rename
            final_obj = context.active_object
            final_obj.name = final_name
            
            # Deselect for next group
            bpy.ops.object.select_all(action='DESELECT')

        # ------------------------------------------------------
        # 4. Cleanup UI
        # ------------------------------------------------------
        
        # Hide Source, Show Target
        source_col.hide_viewport = True
        target_col.hide_viewport = False
        
        self.report({'INFO'}, "Bake Complete. Modifiers Applied & Meshes Joined.")
        return {'FINISHED'}

# ------------------------------------------------------------------------
#    The UI Panel
# ------------------------------------------------------------------------

class VIEW3D_PT_SimpleBakePanel(bpy.types.Panel):
    bl_label = "Collection Baker"
    bl_idname = "VIEW3D_PT_simple_bake"
    bl_space_type = 'VIEW_3D'
    bl_region_type = 'UI'
    bl_category = 'ARAN'

    def draw(self, context):
        layout = self.layout
        scene = context.scene
        props = scene.simple_bake_props

        layout.prop(props, "source_collection")
        layout.prop(props, "target_collection")
        layout.separator()

        if props.source_collection:
            box = layout.box()
            box.label(text="Mesh List:", icon='OUTLINER_OB_MESH')
            
            # Calculate duplicates for display
            name_list = []
            for obj in props.source_collection.objects:
                if obj.type == 'MESH':
                    n = obj.bake_props.target_name.strip()
                    if n: name_list.append(n)
            name_counts = Counter(name_list)

            # Draw List
            col = box.column(align=True)
            for obj in props.source_collection.objects:
                if obj.type != 'MESH':
                    continue
                
                row = col.row(align=True)
                row.label(text=obj.name, icon='MESH_DATA')
                row.prop(obj.bake_props, "target_name", text="")
                
                # Show count if duplicate
                current_target = obj.bake_props.target_name.strip()
                if current_target and name_counts[current_target] > 1:
                    row.label(text=f"{name_counts[current_target]}")

            layout.separator()
            row = layout.row()
            row.scale_y = 1.5
            row.operator("object.simple_bake", icon='RENDER_STILL')
        else:
            layout.label(text="Select Source Collection.", icon='INFO')

# ------------------------------------------------------------------------
#    Registration
# ------------------------------------------------------------------------

classes = (
    BakeObjectProperties,
    BakeSceneProperties,
    OBJECT_OT_SimpleBake,
    VIEW3D_PT_SimpleBakePanel,
)

def register():
    for cls in classes:
        bpy.utils.register_class(cls)
    bpy.types.Scene.simple_bake_props = bpy.props.PointerProperty(type=BakeSceneProperties)
    bpy.types.Object.bake_props = bpy.props.PointerProperty(type=BakeObjectProperties)

def unregister():
    del bpy.types.Scene.simple_bake_props
    del bpy.types.Object.bake_props
    for cls in reversed(classes):
        bpy.utils.unregister_class(cls)

if __name__ == "__main__":
    register()
