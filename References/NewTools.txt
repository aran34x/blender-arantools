# --- Operator for Data Transfer Workflow ---
class RIGHELPER_OT_JoinTargets(bpy.types.Operator):
    """Duplicates, converts, and joins selected objects into a single new mesh"""
    bl_idname = "righelper.join_target_meshes"
    bl_label = "Join Selected Targets"
    bl_description = "1. Select target objects (e.g., armor pieces, clothes).\n2. Click to duplicate, convert to mesh, and join them into a new object.\n3. Originals are hidden"
    bl_options = {'REGISTER', 'UNDO'}

    @classmethod
    def poll(cls, context):
        return context.selected_objects and any(obj.type in {'MESH', 'CURVE', 'SURFACE', 'FONT', 'META'} for obj in context.selected_objects)

    def execute(self, context):
        if context.active_object and context.active_object.mode != 'OBJECT':
            bpy.ops.object.mode_set(mode='OBJECT')

        selected_targets = [obj for obj in context.selected_objects if obj.type in {'MESH', 'CURVE', 'SURFACE', 'FONT', 'META'}]
        if not selected_targets:
            self.report({'ERROR'}, "No convertible objects selected.")
            return {'CANCELLED'}

        originals_to_hide = selected_targets[:]
        bpy.ops.object.duplicate(linked=False)
        for obj in originals_to_hide:
            obj.hide_set(True)
        
        duplicated_objects = context.selected_objects
        if not duplicated_objects:
            self.report({'ERROR'}, "Duplication failed.")
            return {'CANCELLED'}

        context.view_layer.objects.active = duplicated_objects[0]
        try:
            bpy.ops.object.convert(target='MESH')
        except RuntimeError as e:
            self.report({'ERROR'}, f"Conversion to mesh failed: {e}")
            bpy.ops.object.delete()  
            return {'CANCELLED'}

        bpy.ops.object.join()
        joined_mesh = context.active_object
        joined_mesh.name = "Joined_Target"
        
        props = context.scene.adv_rigging_props
        target_collection = props.target_collection
        if target_collection:
            if joined_mesh.name not in target_collection.objects:
                target_collection.objects.link(joined_mesh)
            collections_to_unlink = [coll for coll in joined_mesh.users_collection if coll != target_collection]
            for coll in collections_to_unlink:
                coll.objects.unlink(joined_mesh)

        self.report({'INFO'}, f"Created and joined meshes into: '{joined_mesh.name}'")
        return {'FINISHED'}

class RIGHELPER_OT_BindAndTransfer(bpy.types.Operator):
    """Binds the active mesh to the source's armature and transfers weights"""
    bl_idname = "righelper.bind_and_transfer_weights"
    bl_label = "Bind and Transfer Weights"
    bl_description = "1. Set the character body as 'Source Mesh'.\n2. The new 'Joined_Target' should be the active object.\n3. Click to parent to the source's armature and transfer weights"
    bl_options = {'REGISTER', 'UNDO'}

    @classmethod
    def poll(cls, context):
        props = context.scene.adv_rigging_props
        return props.source_mesh and context.active_object and context.active_object.type == 'MESH'

    def find_armature(self, obj):
        if obj.parent and obj.parent.type == 'ARMATURE':
            return obj.parent
        for mod in obj.modifiers:
            if mod.type == 'ARMATURE' and mod.object:
                return mod.object
        return None

    def execute(self, context):
        props = context.scene.adv_rigging_props
        source_obj = props.source_mesh
        target_obj = context.active_object
        if source_obj == target_obj:
            self.report({'ERROR'}, "Source and Target cannot be the same object.")
            return {'CANCELLED'}

        armature = self.find_armature(source_obj)
        if not armature:
            self.report({'ERROR'}, "Source mesh has no associated armature.")
            return {'CANCELLED'}

        is_posing = (armature.data.pose_position == 'POSE')
        if is_posing:
            armature.data.pose_position = 'REST'

        bpy.ops.object.select_all(action='DESELECT')
        target_obj.select_set(True)
        armature.select_set(True)
        context.view_layer.objects.active = armature
        bpy.ops.object.parent_set(type='ARMATURE_AUTO')
        
        bpy.ops.object.select_all(action='DESELECT')
        target_obj.select_set(True)
        context.view_layer.objects.active = target_obj
        
        mod = target_obj.modifiers.new(name="WeightDataTransfer", type='DATA_TRANSFER')
        mod.object = source_obj
        mod.use_vert_data = True
        mod.data_types_verts = {'VGROUP_WEIGHTS'}
        mod.vert_mapping = props.mapping_method
        bpy.ops.object.modifier_apply(modifier=mod.name)

        if is_posing:
            armature.data.pose_position = 'POSE'
        
        self.report({'INFO'}, f"Weights transferred from '{source_obj.name}' to '{target_obj.name}'")
        return {'FINISHED'}

# --- Operator for Island Weight Copy ---
class RIGHELPER_OT_IslandWeightCopy(bpy.types.Operator):
    """Copies weights from a base vertex to all other vertices in each mesh island"""
    bl_idname = "righelper.island_weight_copy"
    bl_label = "Copy Weights by Island"
    bl_description = "1. For a mesh with disconnected parts (islands), weight one 'base' vertex on each island.\n2. Choose a method to identify the base vertices.\n3. Click to copy the base vertex weights to the rest of its island"
    bl_options = {'REGISTER', 'UNDO'}

    @classmethod
    def poll(cls, context):
        return context.active_object and context.active_object.type == 'MESH'

    def get_islands(self, me):
        """Finds connected vertex islands via a non-recursive search."""
        adjacency = [[] for _ in me.vertices]
        for e in me.edges:
            v1, v2 = e.vertices
            adjacency[v1].append(v2)
            adjacency[v2].append(v1)

        visited = set()
        islands = []
        for v_idx in range(len(me.vertices)):
            if v_idx not in visited:
                stack, island = [v_idx], set()
                visited.add(v_idx)
                while stack:
                    cur = stack.pop()
                    island.add(cur)
                    for neigh in adjacency[cur]:
                        if neigh not in visited:
                            visited.add(neigh)
                            stack.append(neigh)
                islands.append(list(island))
        return islands

    def execute(self, context):
        props = context.scene.adv_rigging_props
        obj = context.active_object
        
        if obj.mode != 'OBJECT':
            bpy.ops.object.mode_set(mode='OBJECT')
        me = obj.data

        islands = self.get_islands(me)
        if not islands:
            self.report({'WARNING'}, "No vertex islands found.")
            return {'CANCELLED'}
        
        selected_verts = {v.index for v in me.vertices if v.select} if props.only_selected_verts else None

        for island in islands:
            source_vert_idx = -1

            if props.base_vertex_method == 'UV_Y':
                if not me.uv_layers.active:
                    self.report({'ERROR'}, "Mesh has no active UV layer for 'Lowest UV-Y' method.")
                    return {'CANCELLED'}
                uv_data = me.uv_layers.active.data
                vert_uvs = {loop.vertex_index: uv_data[loop.index].uv for loop in reversed(me.loops)}
                island_verts_with_uvs = {v_idx for v_idx in island if v_idx in vert_uvs}
                if island_verts_with_uvs:
                    source_vert_idx = min(island_verts_with_uvs, key=lambda v_idx: vert_uvs[v_idx].y)

            elif props.base_vertex_method == 'SHARP':
                sharp_edge_verts = {v for e in me.edges if e.use_edge_sharp for v in e.vertices}
                for v_idx in island:
                    if v_idx in sharp_edge_verts:
                        source_vert_idx = v_idx
                        break
            
            elif props.base_vertex_method == 'ATTRIBUTE':
                attr = me.attributes.get(props.base_attribute_name)
                if not attr or attr.domain != 'POINT' or attr.data_type != 'BOOLEAN':
                    self.report({'ERROR'}, f"Attribute '{props.base_attribute_name}' not found or not a boolean point attribute.")
                    return {'CANCELLED'}
                for v_idx in island:
                    if attr.data[v_idx].value:
                        source_vert_idx = v_idx
                        break
            
            if source_vert_idx == -1:
                continue

            source_weights = {g.group: g.weight for g in me.vertices[source_vert_idx].groups}

            target_verts = island
            if selected_verts is not None:
                target_verts = [v_idx for v_idx in island if v_idx in selected_verts]
            
            if not target_verts:
                continue
                
            if props.blend_factor == 1.0: # FAST PATH
                for vg_idx, weight in source_weights.items():
                    obj.vertex_groups[vg_idx].add(target_verts, weight, 'REPLACE')
                all_vg_indices = set(vg.index for vg in obj.vertex_groups)
                groups_to_remove_from = all_vg_indices - source_weights.keys()
                for vg_idx in groups_to_remove_from:
                    obj.vertex_groups[vg_idx].remove(target_verts)
            else: # SLOW PATH (Blending)
                for v_idx in target_verts:
                    old_weights = {g.group: g.weight for g in me.vertices[v_idx].groups}
                    for vg in obj.vertex_groups:
                        old_w = old_weights.get(vg.index, 0.0)
                        src_w = source_weights.get(vg.index, 0.0)
                        blended_w = old_w * (1 - props.blend_factor) + src_w * props.blend_factor
                        if blended_w > 1e-5:
                            vg.add([v_idx], blended_w, 'REPLACE')
                        elif vg.index in old_weights:
                            vg.remove([v_idx])
        
        self.report({'INFO'}, f"Applied weights to {len(islands)} islands.")
        return {'FINISHED'}

# --- Operator for Weight from Pointer (Auto-Rig Pro) ---
class RIGHELPER_OT_WeightFromPointer(bpy.types.Operator):
    """Iterates through mesh islands, finds the nearest bone tip, and binds a bone chain to each island"""
    bl_idname = "righelper.mesh_island_weight_arp"
    bl_label = "Weight from Pointer"
    bl_description = "For each island, finds the bone tip closest to the pointer, then binds a chain to that island. Repeats for all islands"
    bl_options = {'REGISTER', 'UNDO'}

    def get_islands(self, me):
        """Finds connected vertex islands via a non-recursive search."""
        adjacency = [[] for _ in me.vertices]
        for e in me.edges:
            v1, v2 = e.vertices
            adjacency[v1].append(v2)
            adjacency[v2].append(v1)

        visited = set()
        islands = []
        for v_idx in range(len(me.vertices)):
            if v_idx not in visited:
                stack, island = [v_idx], set()
                visited.add(v_idx)
                while stack:
                    cur = stack.pop()
                    island.add(cur)
                    for neigh in adjacency[cur]:
                        if neigh not in visited:
                            visited.add(neigh)
                            stack.append(neigh)
                islands.append(list(island))
        return islands

    def execute(self, context):
        props = context.scene.adv_rigging_props
        mesh_obj = None
        armature_obj = None

        initial_selection = context.selected_objects[:]
        active_object = context.view_layer.objects.active

        for obj in initial_selection:
            if obj.type == 'MESH':
                mesh_obj = obj
            elif obj.type == 'ARMATURE':
                armature_obj = obj
        
        if not mesh_obj or not armature_obj:
            self.report({'ERROR'}, "Select one Mesh and one Armature.")
            return {'CANCELLED'}
        
        if not hasattr(context.scene, 'arp_bind_selected_bones') or not hasattr(context.scene, 'arp_bind_sel_verts'):
            self.report({'ERROR'}, "Auto-Rig Pro properties not found. Is the addon enabled and active?")
            return {'CANCELLED'}

        if context.active_object and context.active_object.mode != 'OBJECT':
            bpy.ops.object.mode_set(mode='OBJECT')

        mesh_data = mesh_obj.data
        armature_data = armature_obj.data
        
        context.view_layer.objects.active = armature_obj
        bpy.ops.object.mode_set(mode='EDIT')
        # These are the bones the user has selected as potential candidates for the TIP of a chain
        candidate_tip_bone_names = {bone.name for bone in context.selected_bones}
        bpy.ops.object.mode_set(mode='OBJECT')

        if not candidate_tip_bone_names:
            self.report({"ERROR"}, "No bones selected in Armature Edit Mode. Select potential TIP bones for the chains first.")
            return {'CANCELLED'}

        islands = self.get_islands(mesh_data)
        if not islands:
            self.report({'WARNING'}, "No vertex islands found on the mesh.")
            return {'CANCELLED'}
            
        if props.sharp_edge_pointer_method == 'UV_Y_MAX':
            if not mesh_data.uv_layers.active:
                self.report({'ERROR'}, "Mesh has no active UV layer for 'Highest UV-Y' method.")
                return {'CANCELLED'}

        if mesh_obj.vertex_groups:
            context.view_layer.objects.active = mesh_obj
            bpy.ops.object.vertex_group_remove(all=True)

        if not context.scene.arp_bind_selected_bones or not context.scene.arp_bind_sel_verts:
            self.report({'WARNING'}, "'Selected Bones/Verts Only' is disabled in ARP. This may produce unintended results.")

        islands_processed = 0
        for island in islands:
            world_pointer_location = None

            if props.sharp_edge_pointer_method == 'SHARP_EDGE':
                island_verts = set(island)
                found_sharp_edge = None
                for edge in mesh_data.edges:
                    if edge.use_edge_sharp and (edge.vertices[0] in island_verts or edge.vertices[1] in island_verts):
                        found_sharp_edge = edge
                        break
                if found_sharp_edge:
                    v1_co = mesh_data.vertices[found_sharp_edge.vertices[0]].co
                    v2_co = mesh_data.vertices[found_sharp_edge.vertices[1]].co
                    world_pointer_location = mesh_obj.matrix_world @ ((v1_co + v2_co) / 2.0)
            
            elif props.sharp_edge_pointer_method == 'UV_Y_MAX':
                uv_data = mesh_data.uv_layers.active.data
                vert_uvs = {loop.vertex_index: uv_data[loop.index].uv for loop in reversed(mesh_data.loops)}
                island_verts_with_uvs = {v_idx for v_idx in island if v_idx in vert_uvs}
                if island_verts_with_uvs:
                    target_vert_idx = max(island_verts_with_uvs, key=lambda v_idx: vert_uvs[v_idx].y)
                    world_pointer_location = mesh_obj.matrix_world @ mesh_data.vertices[target_vert_idx].co

            if world_pointer_location is None:
                continue

            # Find the best TIP bone for THIS island
            best_tip_bone_name, min_dist = None, float('inf')
            for bone_name in candidate_tip_bone_names:
                rest_bone = armature_obj.data.bones.get(bone_name)
                if rest_bone:
                    # Use the bone's TAIL as the tip
                    bone_tip_world = armature_obj.matrix_world @ rest_bone.tail_local
                    dist = (bone_tip_world - world_pointer_location).length
                    if dist < min_dist:
                        min_dist, best_tip_bone_name = dist, bone_name
            
            if not best_tip_bone_name:
                continue
            
            # --- Perform isolated bind for this island ---
            
            # 1. Select the specific bone chain in the armature
            context.view_layer.objects.active = armature_obj
            bpy.ops.object.mode_set(mode='EDIT')
            bpy.ops.armature.select_all(action='DESELECT')
            
            current_bone = armature_data.edit_bones.get(best_tip_bone_name)
            if current_bone:
                armature_data.edit_bones.active = current_bone
                # Build the chain by selecting parents
                for i in range(props.chain_length):
                    if current_bone:
                        current_bone.select = True
                        current_bone = current_bone.parent
                    else:
                        break
            bpy.ops.object.mode_set(mode='OBJECT')

            # 2. Select the specific island vertices in the mesh
            context.view_layer.objects.active = mesh_obj
            bpy.ops.object.mode_set(mode='EDIT')
            bpy.ops.mesh.select_all(action='DESELECT')
            bpy.ops.object.mode_set(mode='OBJECT') # Must be in object mode to set vertex selections
            for v_idx in island:
                mesh_data.vertices[v_idx].select = True
            
            # 3. Set selections for binding and bind
            bpy.ops.object.select_all(action='DESELECT')
            mesh_obj.select_set(True)
            armature_obj.select_set(True)
            context.view_layer.objects.active = armature_obj
            try:
                bpy.ops.arp.bind_to_rig()
            except AttributeError:
                self.report({'ERROR'}, "'Auto-Rig Pro' is not enabled or installed. Cannot bind.")
                return {'CANCELLED'}
            
            islands_processed += 1
        
        # --- Final Cleanup ---
        bpy.ops.object.mode_set(mode='OBJECT')
        bpy.ops.object.select_all(action='DESELECT')
        for obj in initial_selection:
            obj.select_set(True)
        context.view_layer.objects.active = active_object

        self.report({'INFO'}, f"Weighting complete. Processed {islands_processed} islands.")
        return {'FINISHED'}

# --- Operator for Direct ARP Bind ---
class RIGHELPER_OT_DirectArpBind(bpy.types.Operator):
    """Binds selected vertices to selected bones directly using Auto-Rig Pro"""
    bl_idname = "righelper.direct_arp_bind"
    bl_label = "Direct Bind"
    bl_description = "Binds currently selected vertices to the currently selected bones using Auto-Rig Pro. Assumes you have made the correct selections manually"
    bl_options = {'REGISTER', 'UNDO'}

    @classmethod
    def poll(cls, context):
        # Must have at least one mesh and one armature selected
        sel = context.selected_objects
        return any(o.type == 'MESH' for o in sel) and any(o.type == 'ARMATURE' for o in sel)

    def execute(self, context):
        mesh_obj = None
        armature_obj = None

        # Find the active mesh and armature from the selection
        if context.active_object and context.active_object.type == 'ARMATURE':
            armature_obj = context.active_object
            mesh_obj = next((o for o in context.selected_objects if o.type == 'MESH'), None)
        elif context.active_object and context.active_object.type == 'MESH':
             mesh_obj = context.active_object
             armature_obj = next((o for o in context.selected_objects if o.type == 'ARMATURE'), None)
        else: # Fallback if active object is not mesh or armature
            mesh_obj = next((o for o in context.selected_objects if o.type == 'MESH'), None)
            armature_obj = next((o for o in context.selected_objects if o.type == 'ARMATURE'), None)

        if not mesh_obj or not armature_obj:
            self.report({'ERROR'}, "Select one Mesh and one Armature.")
            return {'CANCELLED'}
        
        if not hasattr(context.scene, 'arp_bind_selected_bones') or not hasattr(context.scene, 'arp_bind_sel_verts'):
            self.report({'ERROR'}, "Auto-Rig Pro properties not found. Is the addon enabled?")
            return {'CANCELLED'}
            
        # Ensure we are in object mode before checking edit mode selections
        if context.mode != 'OBJECT':
             bpy.ops.object.mode_set(mode='OBJECT')

        context.view_layer.objects.active = armature_obj
        bpy.ops.object.mode_set(mode='EDIT')
        selected_bone_count = len(context.selected_bones)
        bpy.ops.object.mode_set(mode='OBJECT')
        
        if selected_bone_count == 0:
            self.report({'ERROR'}, "No bones selected in Armature Edit Mode.")
            return {'CANCELLED'}

        # Final selection state for the operator
        bpy.ops.object.select_all(action='DESELECT')
        mesh_obj.select_set(True)
        armature_obj.select_set(True)
        context.view_layer.objects.active = armature_obj

        try:
            bpy.ops.arp.bind_to_rig()
            self.report({'INFO'}, "Direct bind successful.")
        except AttributeError:
            self.report({'ERROR'}, "'Auto-Rig Pro' addon is not enabled or installed.")
            return {'CANCELLED'}
        except Exception as e:
            self.report({'ERROR'}, f"An error occurred during binding: {e}")
            return {'CANCELLED'}

        context.view_layer.objects.active = mesh_obj

        return {'FINISHED'}
