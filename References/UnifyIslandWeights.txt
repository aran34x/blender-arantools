import bpy
from collections import defaultdict

# ---------------------------------------------------------------------------
# --- 1. PROPERTIES ---
# ---------------------------------------------------------------------------
class UnifyWeightsProperties(bpy.types.PropertyGroup):
    """Properties for the Unify Island Weights Addon."""
    island_unify_blend: bpy.props.FloatProperty(
        name="Blend",
        description="How much to blend towards the unified weight. 0=no change, 1=full unification",
        min=0.0, max=1.0,
        default=1.0
    )
    
    only_selected_verts: bpy.props.BoolProperty(
        name="Only Selected Vertices",
        description="Only apply weights to vertices that are selected in Edit Mode",
        default=False
    )

# ---------------------------------------------------------------------------
# --- 2. OPERATOR ---
# ---------------------------------------------------------------------------
class UNIFYWEIGHTS_OT_Execute(bpy.types.Operator):
    """For each island, finds the bone with the most influence and assigns the entire island to it"""
    bl_idname = "unifyweights.execute"
    bl_label = "Unify Island Weights"
    bl_description = ("1. Run after an automatic weighting.\n"
                      "2. For each separate mesh island, finds the bone with the most influence.\n"
                      "3. Blends the weights of the entire island to be 100% influenced by that single bone")
    bl_options = {'REGISTER', 'UNDO'}

    @classmethod
    def poll(cls, context):
        obj = context.active_object
        return obj and obj.type == 'MESH' and obj.vertex_groups

    def get_islands(self, me):
        """Finds connected vertex islands via a non-recursive search."""
        adjacency = [[] for _ in me.vertices]
        for e in me.edges:
            v1, v2 = e.vertices
            adjacency[v1].append(v2)
            adjacency[v2].append(v1)

        visited = set()
        islands = []
        for v_idx in range(len(me.vertices)):
            if v_idx not in visited:
                stack, island = [v_idx], set()
                visited.add(v_idx)
                while stack:
                    cur = stack.pop()
                    island.add(cur)
                    for neigh in adjacency[cur]:
                        if neigh not in visited:
                            visited.add(neigh)
                            stack.append(neigh)
                islands.append(list(island))
        return islands

    def execute(self, context):
        props = context.scene.unify_weights_props
        obj = context.active_object
        
        if obj.mode != 'OBJECT':
            bpy.ops.object.mode_set(mode='OBJECT')
        me = obj.data
        blend_factor = props.island_unify_blend

        selected_verts = None
        if props.only_selected_verts:
            selected_verts = {v.index for v in me.vertices if v.select}
            if not selected_verts:
                self.report({'WARNING'}, "Option 'Only Selected' is on, but no vertices are selected.")
                return {'CANCELLED'}

        islands = self.get_islands(me)
        
        if not islands:
            self.report({'WARNING'}, "No vertex islands found.")
            return {'CANCELLED'}

        islands_processed = 0
        for island in islands:
            if not island:
                continue

            # Calculate the total weight for each group on this island (always based on the whole island)
            group_total_weights = defaultdict(float)
            for v_idx in island:
                for group_element in me.vertices[v_idx].groups:
                    group_total_weights[group_element.group] += group_element.weight

            if not group_total_weights:
                continue  # Skip islands with no weights

            # Find the group with the highest total weight
            winning_group_idx = max(group_total_weights, key=group_total_weights.get)

            # Determine which vertices to actually modify
            target_verts = island
            if selected_verts is not None:
                # Intersect island vertices with the user's selection
                target_verts = [v_idx for v_idx in island if v_idx in selected_verts]

            if not target_verts:
                continue

            if blend_factor == 1.0:
                # FAST PATH: Batch process all vertices at once
                groups_to_clear = set()
                for v_idx in target_verts:
                    for group in me.vertices[v_idx].groups:
                        groups_to_clear.add(group.group)
                
                if winning_group_idx in groups_to_clear:
                    groups_to_clear.remove(winning_group_idx)
                
                # Batch remove from all non-winning groups
                for vg_idx in groups_to_clear:
                    obj.vertex_groups[vg_idx].remove(target_verts)
                
                # Batch add to the winning group
                obj.vertex_groups[winning_group_idx].add(target_verts, 1.0, 'REPLACE')
            else:
                # --- HEAVILY OPTIMIZED BLEND PATH ---
                # This approach is much faster as it avoids per-vertex API calls by batching per-group.
                
                # 1. Cache all initial weights for the target vertices in one go.
                initial_weights = {v_idx: {g.group: g.weight for g in me.vertices[v_idx].groups} for v_idx in target_verts}
                
                # 2. Determine the full set of vertex groups we need to modify.
                all_affected_groups = set()
                all_affected_groups.add(winning_group_idx)
                for v_weights in initial_weights.values():
                    all_affected_groups.update(v_weights.keys())

                # 3. Iterate over each GROUP, not each vertex.
                for vg_idx in all_affected_groups:
                    vg = obj.vertex_groups[vg_idx]
                    is_winner = (vg_idx == winning_group_idx)
                    
                    verts_to_remove = []
                    # Group vertices by their final calculated weight to enable batching.
                    verts_to_add = defaultdict(list)

                    # 4. Calculate the final weights for all vertices for THIS group.
                    for v_idx in target_verts:
                        old_w = initial_weights.get(v_idx, {}).get(vg_idx, 0.0)
                        target_w = 1.0 if is_winner else 0.0
                        blended_w = old_w * (1.0 - blend_factor) + target_w * blend_factor

                        if blended_w > 1e-5:
                            # Rounding allows us to group verts with similar weights into a single .add() call
                            weight_key = round(blended_w, 4)
                            verts_to_add[weight_key].append(v_idx)
                        elif old_w > 1e-5: # Only remove if it had weight before
                            verts_to_remove.append(v_idx)

                    # 5. Apply all changes for this group in a few batched API calls.
                    if verts_to_remove:
                        vg.remove(verts_to_remove)
                    
                    for weight, verts in verts_to_add.items():
                        if verts:
                            vg.add(verts, weight, 'REPLACE')
            
            islands_processed += 1

        self.report({'INFO'}, f"Unified weights for {islands_processed} islands.")
        return {'FINISHED'}

# ---------------------------------------------------------------------------
# --- 3. UI PANEL ---
# ---------------------------------------------------------------------------
class VIEW3D_PT_UnifyIslandWeightsPanel(bpy.types.Panel):
    bl_space_type = 'VIEW_3D'
    bl_region_type = 'UI'
    bl_category = 'Rigging Helper'
    bl_label = "Unify Island Weights"

    def draw(self, context):
        layout = self.layout
        props = context.scene.unify_weights_props

        box = layout.box()
        col = box.column(align=True)
        col.prop(props, "island_unify_blend")
        col.prop(props, "only_selected_verts")
        col.operator(UNIFYWEIGHTS_OT_Execute.bl_idname, text="Unify Weights")

# ---------------------------------------------------------------------------
# --- 4. REGISTRATION ---
# ---------------------------------------------------------------------------
classes = (
    UnifyWeightsProperties,
    UNIFYWEIGHTS_OT_Execute,
    VIEW3D_PT_UnifyIslandWeightsPanel,
)

def register():
    for cls in classes:
        bpy.utils.register_class(cls)
    bpy.types.Scene.unify_weights_props = bpy.props.PointerProperty(type=UnifyWeightsProperties)

def unregister():
    del bpy.types.Scene.unify_weights_props
    for cls in reversed(classes):
        bpy.utils.unregister_class(cls)

if __name__ == "__main__":
    register()
