import bpy

class OBJECT_OT_smart_weight_transfer(bpy.types.Operator):
    """Transfer Weights + Bind to Armature"""
    bl_idname = "object.smart_weight_transfer"
    bl_label = "Transfer & Bind"
    bl_options = {'REGISTER', 'UNDO'}

    def execute(self, context):
        props = context.scene.weight_transfer_props
        source = props.source_mesh
        targets = [obj for obj in context.selected_objects if obj.type == 'MESH']

        # --- CHECKS ---
        if not source:
            self.report({'ERROR'}, "Assign a SOURCE mesh in the panel!")
            return {'CANCELLED'}
        
        if len(source.vertex_groups) == 0:
            self.report({'ERROR'}, f"Source '{source.name}' has no vertex groups to transfer!")
            return {'CANCELLED'}

        if source in targets:
            targets.remove(source)

        if not targets:
            self.report({'ERROR'}, "Select TARGET meshes in the 3D Viewport!")
            return {'CANCELLED'}

        # Find Armature
        armature = source.parent
        if not armature or armature.type != 'ARMATURE':
            for mod in source.modifiers:
                if mod.type == 'ARMATURE' and mod.object:
                    armature = mod.object
                    break
        
        if not armature:
            self.report({'ERROR'}, f"Source '{source.name}' is not rigged (No Armature found)!")
            return {'CANCELLED'}

        count = 0
        
        # Ensure Object Mode
        if bpy.context.object and bpy.context.object.mode != 'OBJECT':
            bpy.ops.object.mode_set(mode='OBJECT')

        for target in targets:
            # Shape Key Check
            if target.data.shape_keys:
                self.report({'ERROR'}, f"Skipping '{target.name}': Has Shape Keys.")
                continue

            context.view_layer.objects.active = target
            
            # --- STEP 0: CREATE VERTEX GROUPS (The Fix) ---
            # We manually create the groups first to ensure they exist
            for src_grp in source.vertex_groups:
                if src_grp.name not in target.vertex_groups:
                    target.vertex_groups.new(name=src_grp.name)
            
            # --- STEP 1: DATA TRANSFER MODIFIER ---
            dt_mod = target.modifiers.new(name="WeightTransfer_Temp", type='DATA_TRANSFER')
            dt_mod.object = source
            dt_mod.use_vert_data = True
            dt_mod.data_types_verts = {'VGROUP_WEIGHTS'}
            dt_mod.vert_mapping = props.interp_mode
            
            if props.interp_mode == 'NEAREST_POLYNOR':
                dt_mod.use_max_distance = False

            # APPLY
            try:
                bpy.ops.object.modifier_apply(modifier=dt_mod.name)
            except RuntimeError as e:
                self.report({'ERROR'}, f"Modifier Apply failed on {target.name}. See Console.")
                print(f"Error: {e}")
                continue
            
            # --- STEP 2: CLEAN ---
            # Now safe to run because groups were pre-created
            if props.clean_empty:
                # Only run if groups actually exist (double check)
                if len(target.vertex_groups) > 0:
                    bpy.ops.object.vertex_group_clean(group_select_mode='ALL', limit=0.001)

            # --- STEP 3: BIND TO ARMATURE ---
            if target.parent != armature:
                bpy.ops.object.select_all(action='DESELECT')
                target.select_set(True)
                armature.select_set(True)
                context.view_layer.objects.active = armature
                bpy.ops.object.parent_set(type='ARMATURE', keep_transform=True)
            
            # --- STEP 4: ENSURE MODIFIER ---
            arm_mod = None
            for mod in target.modifiers:
                if mod.type == 'ARMATURE':
                    arm_mod = mod
                    break
            
            if not arm_mod:
                arm_mod = target.modifiers.new(name="Armature", type='ARMATURE')
                arm_mod.object = armature
            
            arm_mod.show_viewport = True
            count += 1

        if count > 0:
            self.report({'INFO'}, f"Success! Transferred & Bound {count} meshes.")
        else:
            self.report({'WARNING'}, "No meshes were bound. Check errors.")
            
        return {'FINISHED'}

class WeightTransferProperties(bpy.types.PropertyGroup):
    source_mesh: bpy.props.PointerProperty(
        type=bpy.types.Object,
        name="Source",
        description="The rigged mesh to copy from",
        poll=lambda self, obj: obj.type == 'MESH'
    )
    
    interp_mode: bpy.props.EnumProperty(
        name="Method",
        items=[
            ('NEAREST', "Nearest Vertex", "Fastest"),
            ('POLYINTERP_NEAREST', "Nearest Face (Smooth)", "Standard"),
            ('POLYINTERP_VNORPROJ', "Projected Face", "Raycast"),
        ],
        default='POLYINTERP_NEAREST'
    )
    
    clean_empty: bpy.props.BoolProperty(name="Clean Empty Groups", default=True)

class VIEW3D_PT_smart_transfer(bpy.types.Panel):
    bl_space_type = 'VIEW_3D'
    bl_region_type = 'UI'
    bl_category = "Rigging"
    bl_label = "Smart Weight Transfer"
    
    def draw(self, context):
        layout = self.layout
        props = context.scene.weight_transfer_props
        layout.prop(props, "source_mesh")
        box = layout.box()
        box.label(text="Transfer Settings")
        box.prop(props, "interp_mode", text="")
        box.prop(props, "clean_empty")
        layout.separator()
        col = layout.column()
        col.scale_y = 1.6
        col.operator("object.smart_weight_transfer", icon='ARMATURE_DATA')

def register():
    bpy.utils.register_class(WeightTransferProperties)
    bpy.utils.register_class(OBJECT_OT_smart_weight_transfer)
    bpy.utils.register_class(VIEW3D_PT_smart_transfer)
    bpy.types.Scene.weight_transfer_props = bpy.props.PointerProperty(type=WeightTransferProperties)

def unregister():
    bpy.utils.unregister_class(WeightTransferProperties)
    bpy.utils.unregister_class(OBJECT_OT_smart_weight_transfer)
    bpy.utils.unregister_class(VIEW3D_PT_smart_transfer)
    del bpy.types.Scene.weight_transfer_props

if __name__ == "__main__":
    register()
