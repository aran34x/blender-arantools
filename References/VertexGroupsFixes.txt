import bpy

# ---------------------------------------------------------------------------
# --- 1. OPERATOR ---
# ---------------------------------------------------------------------------
class SYNCGROUPS_OT_Execute(bpy.types.Operator):
    """Adds vertex groups to the active mesh to match the deform bones of its armature"""
    bl_idname = "righelper.sync_vertex_groups"
    bl_label = "Sync Deform Bone Groups"
    bl_description = "Adds empty vertex groups for any deform bones in the armature that don't have a group on the active mesh"
    bl_options = {'REGISTER', 'UNDO'}

    @classmethod
    def poll(cls, context):
        # This operator is only active if the selected object is a mesh
        obj = context.active_object
        return obj and obj.type == 'MESH'

    def find_armature(self, obj):
        """Finds the armature associated with a mesh object."""
        # Check for an armature modifier first, as it's most definitive
        for mod in obj.modifiers:
            if mod.type == 'ARMATURE' and mod.object:
                return mod.object
        # As a fallback, check if the object is parented to an armature
        if obj.parent and obj.parent.type == 'ARMATURE':
            return obj.parent
        return None

    def execute(self, context):
        obj = context.active_object
        if not obj:
            self.report({'ERROR'}, "No active mesh object selected.")
            return {'CANCELLED'}

        armature_obj = self.find_armature(obj)
        if not armature_obj:
            self.report({'ERROR'}, "Could not find an associated armature for the active mesh.")
            return {'CANCELLED'}

        # Get a set of all deform bone names from the armature
        deform_bone_names = {bone.name for bone in armature_obj.data.bones if bone.use_deform}

        # Get a set of all existing vertex group names from the mesh
        existing_group_names = {vg.name for vg in obj.vertex_groups}

        # Find which bone names are missing from the vertex groups
        missing_names = deform_bone_names - existing_group_names

        if not missing_names:
            self.report({'INFO'}, "Vertex groups are already in sync with deform bones.")
            return {'FINISHED'}

        # Create a new, empty vertex group for each missing name
        # Sorted alphabetically to ensure a predictable order in the UI
        for name in sorted(list(missing_names)):
            obj.vertex_groups.new(name=name)

        self.report({'INFO'}, f"Added {len(missing_names)} missing vertex groups.")
        return {'FINISHED'}

# ---------------------------------------------------------------------------
# --- 2. UI PANEL ---
# ---------------------------------------------------------------------------
class VIEW3D_PT_SyncVertexGroupsPanel(bpy.types.Panel):
    bl_space_type = 'VIEW_3D'
    bl_region_type = 'UI'
    bl_category = 'Rigging Helper'  # Uses the same category to appear in the same tab
    bl_label = "Sync Vertex Groups"
    bl_options = {'DEFAULT_CLOSED'} # Starts closed to save space

    def draw(self, context):
        layout = self.layout
        box = layout.box()
        col = box.column()
        col.operator(SYNCGROUPS_OT_Execute.bl_idname, text="Sync Groups from Armature")

# ---------------------------------------------------------------------------
# --- 3. REGISTRATION ---
# ---------------------------------------------------------------------------
classes = (
    SYNCGROUPS_OT_Execute,
    VIEW3D_PT_SyncVertexGroupsPanel,
)

def register():
    for cls in classes:
        bpy.utils.register_class(cls)

def unregister():
    for cls in reversed(classes):
        bpy.utils.unregister_class(cls)

if __name__ == "__main__":
    register()
